{%- comment -%} Site override of Hydejack's breadcrumbs. The stock include prints the raw URL segments ("rfidtoolstorage", "E1", "movingpath"); this one looks each parent URL up and prints that page's `crumb` (or title), so the trail reads like the navigation does. Segments with no page behind them (for example /tools/) are shown as plain text. A page can set `crumb_root` (a site URL) to put that page in the trail in place of the first URL segment; the Rocket League pages use it to sit under Notes. {%- endcomment -%} {%- assign bc_url = include.url | remove: '/index.html' -%} {%- assign bc_segs = bc_url | split: '/' -%} {%- if bc_segs.size > 1 and page.hide_breadcrumbs != true -%}
  • Home
    • {%- assign bc_path = '' -%} {%- if page.crumb_root -%} {%- assign bc_root = site.pages | where: "url", page.crumb_root | first -%}
  • / {{ bc_root.crumb | default: bc_root.title }}
    • {%- endif -%} {%- for seg in bc_segs offset: 1 -%} {%- assign bc_path = bc_path | append: '/' | append: seg -%} {%- assign bc_href = bc_path | append: '/' -%} {%- assign bc_page = site.pages | where: "url", bc_href | first -%} {%- if page.crumb_root and forloop.first -%} {%- continue -%} {%- endif -%} {%- if forloop.last -%}
  • / {{ bc_page.crumb | default: bc_page.title | default: seg | url_decode }}
    • {%- elsif bc_page -%}
  • / {{ bc_page.crumb | default: bc_page.title }}
    • {%- endif -%} {%- endfor -%}
{%- endif -%}
